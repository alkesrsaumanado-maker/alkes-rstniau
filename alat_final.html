// ============================================================ // SISTEM MONITORING ALAT KESEHATAN — RS TNI AU // Google Apps Script | Versi 2 (CORS Fix) // ============================================================ var SPREADSHEET_ID = "1WWilWzD2RS3EWPv5sP4NZS7UcM-7Dev2tzyL9qfzLrA"; var SHEET_DATABASE = "📋 Database Alkes"; var SHEET_RIWAYAT = "📝 Riwayat Pemeliharaan"; // ============================================================ // FUNGSI UTAMA — semua request lewat doGet (GET saja) // ============================================================ function doGet(e) { var action = e.parameter.action || "getAlat"; var kode = e.parameter.kode || ""; var result; if (action === "getAlat") { result = getDataAlat(kode); } else if (action === "getRiwayat") { result = getRiwayatAlat(kode); } else if (action === "tambahRiwayat") { result = tambahRiwayatDariGet(e.parameter); } else if (action === "updateStatus") { result = updateStatusAlat(e.parameter); } else { result = { error: "Action tidak dikenal" }; } // CORS header lewat callback/JSONP tidak diperlukan — // cukup kembalikan JSON biasa, Apps Script sudah handle CORS untuk GET var output = ContentService .createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); return output; } // ============================================================ // BACA DATA ALAT // ============================================================ function getDataAlat(kode) { try { var ss = SpreadsheetApp.openById(SPREADSHEET_ID); var sheet = ss.getSheetByName(SHEET_DATABASE); var data = sheet.getDataRange().getValues(); for (var i = 3; i < data.length; i++) { var row = data[i]; if (String(row[1]).trim() === kode.trim()) { var kalibBerikutnya = row[11] ? new Date(row[11]) : null; var sisaHari = kalibBerikutnya ? Math.round((kalibBerikutnya - new Date()) / (1000 * 60 * 60 * 24)) : null; return { no : row[0], kode : row[1], nama : row[2], kategori : row[3], merek : row[4], nomorSeri : row[5], ruangan : row[6], penanggungjawab: row[7], tahunPengadaan : row[8], kondisi : row[9], kalibTerakhir : row[10] ? Utilities.formatDate(new Date(row[10]), "Asia/Jakarta", "dd MMM yyyy") : "-", kalibBerikutnya: kalibBerikutnya ? Utilities.formatDate(kalibBerikutnya, "Asia/Jakarta", "dd MMM yyyy") : "-", sisaHari : sisaHari, statusKalibrasi: row[13], catatan : row[15], diperbarui : row[16] ? Utilities.formatDate(new Date(row[16]), "Asia/Jakarta", "dd MMM yyyy") : "-" }; } } return { error: "Alat dengan kode " + kode + " tidak ditemukan" }; } catch (err) { return { error: err.message }; } } // ============================================================ // BACA RIWAYAT PEMELIHARAAN // ============================================================ function getRiwayatAlat(kode) { try { var ss = SpreadsheetApp.openById(SPREADSHEET_ID); var sheet = ss.getSheetByName(SHEET_RIWAYAT); var data = sheet.getDataRange().getValues(); var hasil = []; for (var i = 3; i < data.length; i++) { var row = data[i]; if (String(row[1]).trim() === kode.trim() && row[1] !== "") { hasil.push({ tanggal : row[3] ? Utilities.formatDate(new Date(row[3]), "Asia/Jakarta", "dd MMM yyyy") : "-", jenis : row[4], pelaksana : row[5], hasil : row[6], biaya : row[7] ? Number(row[7]).toLocaleString("id-ID") : "0", dicatatOleh: row[9] }); } } hasil.sort(function(a, b) { return new Date(b.tanggal) - new Date(a.tanggal); }); return { riwayat: hasil }; } catch (err) { return { error: err.message }; } } // ============================================================ // TAMBAH RIWAYAT — via GET parameter // ============================================================ function tambahRiwayatDariGet(p) { try { var ss = SpreadsheetApp.openById(SPREADSHEET_ID); var sheet = ss.getSheetByName(SHEET_RIWAYAT); var lastRow = sheet.getLastRow(); var newRow = lastRow + 1; var no = lastRow - 2; var tanggal = new Date(p.tanggal); sheet.getRange(newRow, 1).setValue(no); sheet.getRange(newRow, 2).setValue(p.kode); sheet.getRange(newRow, 3).setValue(p.namaAlat); sheet.getRange(newRow, 4).setValue(tanggal); sheet.getRange(newRow, 4).setNumberFormat("DD MMM YYYY"); sheet.getRange(newRow, 5).setValue(p.jenis); sheet.getRange(newRow, 6).setValue(p.pelaksana); sheet.getRange(newRow, 7).setValue(p.keterangan); sheet.getRange(newRow, 8).setValue(p.biaya ? Number(p.biaya) : 0); sheet.getRange(newRow, 9).setValue(p.dokumen || ""); sheet.getRange(newRow, 10).setValue(p.dicatatOleh); if (p.jenis === "Kalibrasi Berkala" && p.kalibBerikutnya) { updateKalibrasi(p.kode, tanggal, new Date(p.kalibBerikutnya)); } return { sukses: true, pesan: "Data berhasil disimpan ke Google Sheets!" }; } catch (err) { return { sukses: false, error: err.message }; } } // ============================================================ // UPDATE STATUS ALAT // ============================================================ function updateStatusAlat(p) { try { var ss = SpreadsheetApp.openById(SPREADSHEET_ID); var sheet = ss.getSheetByName(SHEET_DATABASE); var rows = sheet.getDataRange().getValues(); for (var i = 3; i < rows.length; i++) { if (String(rows[i][1]).trim() === p.kode.trim()) { sheet.getRange(i + 1, 10).setValue(p.kondisi); sheet.getRange(i + 1, 16).setValue(p.catatan); sheet.getRange(i + 1, 17).setValue(new Date()); return { sukses: true, pesan: "Status alat berhasil diperbarui!" }; } } return { sukses: false, error: "Kode alat tidak ditemukan" }; } catch (err) { return { sukses: false, error: err.message }; } } // ============================================================ // UPDATE TANGGAL KALIBRASI // ============================================================ function updateKalibrasi(kode, kalibTerakhir, kalibBerikutnya) { try { var ss = SpreadsheetApp.openById(SPREADSHEET_ID); var sheet = ss.getSheetByName(SHEET_DATABASE); var data = sheet.getDataRange().getValues(); for (var i = 3; i < data.length; i++) { if (String(data[i][1]).trim() === kode.trim()) { sheet.getRange(i + 1, 11).setValue(kalibTerakhir); sheet.getRange(i + 1, 11).setNumberFormat("DD MMM YYYY"); sheet.getRange(i + 1, 12).setValue(kalibBerikutnya); sheet.getRange(i + 1, 12).setNumberFormat("DD MMM YYYY"); sheet.getRange(i + 1, 17).setValue(new Date()); break; } } } catch (err) { Logger.log("Error update kalibrasi: " + err.message); } }
